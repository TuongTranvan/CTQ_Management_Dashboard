# app/database/register_tables.py
from app.database.connection import conn
from app.config.reflow_deflux_line_machine_config import MACHINE_MAPPING


def init_duckdb_views():
    # 1. Macro bóc tách thời gian 14 số (YYYYMMDDHHMMSS) từ Source_File, bù bằng cột thời gian gốc nếu thiếu
    conn.execute("""
    CREATE OR REPLACE MACRO get_effective_time(fname, fallback_time) AS 
        COALESCE(
            try_strptime(regexp_extract(CAST(fname AS VARCHAR), '(?:^|_)(\\d{14})(?:_|\\.|$)', 1), '%Y%m%d%H%M%S'),
            try_cast(fallback_time AS TIMESTAMP)
        );
    """)

    # 2. Macro bóc tách Serial (VD: 2025-1100019) từ Source_File (bất chấp vị trí dấu '_')
    conn.execute("""
    CREATE OR REPLACE MACRO get_file_serial(fname) AS 
        regexp_extract(CAST(fname AS VARCHAR), '(?:^|_)(\\d{4}-\\d{5,10})(?:_|\\.|$)', 1);
    """)

    # 3. Macro bóc tách tag Deflux1/2/3 từ Source_File
    conn.execute("""
    CREATE OR REPLACE MACRO get_deflux_tag(fname) AS 
        COALESCE(
            NULLIF(regexp_extract(CAST(fname AS VARCHAR), '(?:^|_)(Deflux\\d+)(?:_|\\.|$)', 1), ''),
            'Unknown'
        );
    """)

    # 4. Tự động sinh câu lệnh CASE WHEN từ MACHINE_MAPPING để map chuẩn Tên máy, Line, Station
    name_cases = []
    line_cases = []
    station_cases = []

    for m_name, info in MACHINE_MAPPING.items():
        serial = info["machine_id"]
        cond = f"(Machine_ID = '{m_name}' OR Machine_ID = '{serial}' OR get_file_serial(Source_File) = '{serial}')"
        
        name_cases.append(f"WHEN {cond} THEN '{m_name}'")

        if info["type"] == "Reflow":
            line_cases.append(f"WHEN {cond} THEN '{info['line']}'")
            station_cases.append(f"WHEN {cond} THEN '{info['station']}'")
        else:
            # Với Deflux: dựa thêm vào tag Deflux1/2/3 trên Source_File để tìm đúng station (964, 965, 966)
            for st_code, st_line in info["stations"].items():
                deflux_cond = f"{cond} AND LOWER(get_deflux_tag(Source_File)) = LOWER('{st_line}')"
                line_cases.append(f"WHEN {deflux_cond} THEN '{st_line}'")
                station_cases.append(f"WHEN {deflux_cond} THEN '{st_code}'")

    name_sql = f"CASE {' '.join(name_cases)} ELSE COALESCE(Machine_ID, 'Unknown') END"
    line_sql = f"CASE {' '.join(line_cases)} ELSE get_deflux_tag(Source_File) END"
    station_sql = f"CASE {' '.join(station_cases)} ELSE 'Unknown' END"

# ======================
# UPLOAD RATE and cards
# ======================

    # 5. Tạo View cho Reflow (chỉ đọc các cột cần thiết từ Parquet) dùng cho card và upload rate
    conn.execute(f"""
    CREATE OR REPLACE VIEW reflow_ctq AS 
    SELECT
        Panel_SN,
        {name_sql} AS machine_name,
        {line_sql} AS line,
        {station_sql} AS station,
        get_effective_time(Source_File, StartTime) AS eff_time,
        Source_Folder
    FROM read_parquet('data/reflow_ctq.parquet');
    """)

    # 6. Tạo View cho Deflux (chỉ đọc các cột cần thiết từ Parquet)dùng cho card và upload rate
    conn.execute(f"""
    CREATE OR REPLACE VIEW deflux_ctq AS 
    SELECT 
        Panel_SN,
        {name_sql} AS machine_name,
        {line_sql} AS line,
        {station_sql} AS station,
        get_effective_time(Source_File, START_TIME) AS eff_time,
        Source_Folder
    FROM read_parquet('data/deflux_ctq.parquet');
    """)

# ======================
# Spec rate upload success full 
# ======================

    #7. Tạo view reflow cho phần out spec rate
    conn.execute(f"""
    CREATE OR REPLACE VIEW reflow_ctq_fail AS
    SELECT
        *,
        {name_sql} AS machine_name,
        {line_sql} AS line,
        {station_sql} AS station,
        get_effective_time(Source_File, StartTime) AS eff_time
    FROM read_parquet('data/reflow_ctq.parquet')
    WHERE
        Source_Folder LIKE '%CTQ_FAIL_SPEC%'
        OR
        Source_Folder LIKE '%CTQ_FAIL_COV_AND_SPEC%';
    """)


    #8. Tạo view deflux cho phần out spec rate
    conn.execute(f"""
    create or replace view deflux_ctq_fail as
    select *
    from read_parquet('data/deflux_ctq.parquet')
    where
        Source_Folder like '%CTQ_FAIL_SPEC%' or Source_Folder like '%CTQ_FAIL_COV_AND_SPEC%'
    """)


# ======================
# Out of spec list
# ======================

    #9. tạo view cho phần Reflow Out Spec List
    conn.execute(f"""
    CREATE OR REPLACE VIEW reflow_ctq_analysis AS
    SELECT
        *,
        {name_sql} AS machine_name,
        {line_sql} AS line,
        {station_sql} AS station,
        get_effective_time(Source_File, StartTime) AS eff_time
    FROM read_parquet('data/reflow_ctq.parquet')
    WHERE
        Source_Folder LIKE '%CTQ_FAIL_SPEC%'
        OR Source_Folder LIKE '%CTQ_FAIL_COV%'
        OR Source_Folder LIKE '%CTQ_FAIL_COV_AND_SPEC%';
    """)

    #10.tạo view cho phần Deflux Out Spec List
    conn.execute(f"""
    CREATE OR REPLACE VIEW deflux_ctq_analysis AS
    SELECT
        *,
        {name_sql} AS machine_name,
        {line_sql} AS line,
        {station_sql} AS station,
        get_effective_time(Source_File, START_TIME) AS eff_time
    FROM read_parquet('data/deflux_ctq.parquet')
    WHERE
        Source_Folder LIKE '%CTQ_FAIL_SPEC%'
        OR Source_Folder LIKE '%CTQ_FAIL_COV%'
        OR Source_Folder LIKE '%CTQ_FAIL_COV_AND_SPEC%';
    """)






# Tự động chạy khi module được import
init_duckdb_views()
