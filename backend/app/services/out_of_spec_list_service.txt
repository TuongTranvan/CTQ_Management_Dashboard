import calendar
from datetime import date, datetime, timedelta

import pandas as pd

from app.config.cov_config import COV_DEFLUX, COV_REFLOW
from app.repository.out_of_spec_list_repository import OutOfSpecListRepository


class OutOfSpecListService:
    @staticmethod
    def build_where_clause(
        process: str,
        period_type: str,
        period_value: str,
        scope: str = "All"
    ):
        conditions = ["eff_time IS NOT NULL"]
        p_val = period_value.strip()

        if period_type.lower() == "month":
            conditions.append(
                f"strftime(eff_time, '%Y-%m') = '{p_val}'"
            )

        elif period_type.lower() == "week":

            if "-W" in p_val.upper():

                conditions.append(
                    f"strftime(eff_time, '%G-W%V') = '{p_val.upper()}'"
                )

            else:

                curr_year = datetime.now().year

                week_num = (
                    p_val.upper()
                    .replace("W", "")
                    .zfill(2)
                )

                conditions.append(
                    f"strftime(eff_time, '%G-W%V') = '{curr_year}-W{week_num}'"
                )

        if (
            scope
            and scope.strip().lower() != "all"
        ):
            conditions.append(
                f"LOWER(line) = LOWER('{scope}')"
            )

        return (
            "WHERE "
            + " AND ".join(conditions)
        )

    @staticmethod
    def get_cov_limit(parameter):
        p = parameter.upper()

        if "E_ZONE11" in p and "SENSOR" in p:
            return 6

        if (p.startswith("B_") or p.startswith("E_")) and "SENSOR" in p:
            return 5

        if "FAN_SPEED" in p:
            return 5

        if "CHAIN_SPEED" in p:
            return 5

        if "REFLOW_TIME" in p:
            return 5

        if "PRESSURE" in p:
            return 8

        if "TEMP" in p:
            return 5

        return None

    @staticmethod
    def calculate_cov(values):
        mean_value = values.mean()

        if mean_value == 0:
            return 0

        stdev_value = values.std()

        return (stdev_value / mean_value) * 100

    @staticmethod
    def is_cov_fail(parameter, cov_value):
        p = parameter.upper()

        if cov_value == 0:
            if (
                "OXYGEN" in p
                or "REFLOW_TIME" in p
                or "FAN_SPEED" in p
                or "CHAIN_SPEED" in p
                or "PRESSURE" in p
                or "TEMP" in p
                or "CONCENTRATION" in p
                or "DI_WATER" in p
            ):
                return True

        limit = OutOfSpecListService.get_cov_limit(parameter)

        if limit is not None and cov_value > limit:
            return True

        return False

    @staticmethod
    def build_out_spec_rows(df, parameter_defs):
        rows = []

        for parameter_info in parameter_defs:
            parameter = parameter_info["parameter"]
            value_col = parameter_info["value_col"]
            result_col = parameter_info["result_col"]
            upper_col = parameter_info["upper_col"]
            lower_col = parameter_info["lower_col"]

            # Bỏ qua nếu thiếu cột
            if result_col not in df.columns:
                continue

            fail_df = df[
                df[result_col].astype(str).str.upper().eq("FAIL")
            ]

            if fail_df.empty:
                continue

            # Group theo line
            for line, line_df in fail_df.groupby("line"):
                # Avg value
                avg_value = None

                if value_col in line_df.columns:

                    value_series = pd.to_numeric(
                        line_df[value_col],
                        errors="coerce"
                    )

                    if value_series.notna().any():

                        avg_value = round(
                            value_series.mean(),
                            4
                        )

                # Avg usl
                avg_usl = None

                if upper_col in line_df.columns:

                    usl_series = pd.to_numeric(
                        line_df[upper_col],
                        errors="coerce"
                    )

                    if usl_series.notna().any():

                        avg_usl = round(
                            usl_series.mean(),
                            4
                        )

                # Avg lsl
                avg_lsl = None

                if lower_col in line_df.columns:

                    lsl_series = pd.to_numeric(
                        line_df[lower_col],
                        errors="coerce"
                    )

                    if lsl_series.notna().any():

                        avg_lsl = round(
                            lsl_series.mean(),
                            4
                        )

                # Fail strip count
                strip_count = line_df["Panel_SN"].nunique()

                # Top 3 machine
                machine_rank = (
                    line_df.groupby("machine_name")["Panel_SN"]
                    .nunique()
                    .sort_values(ascending=False)
                    .head(3)
                )
                top3_machine = list(machine_rank.index)

                rows.append(
                    {
                        "line": line,
                        "parameter": parameter,
                        "top3_machine": ", ".join(top3_machine),
                        "value": avg_value,
                        "usl": avg_usl,
                        "lsl": avg_lsl,
                        "number_of_strip": int(strip_count),
                        "status": "Out Spec",
                    }
                )

        return rows


    @staticmethod
    def build_out_cov_rows(
        df,
        process: str
    ):

        rows = []

        cov_parameters = (
            COV_REFLOW
            if process.lower() == "reflow"
            else COV_DEFLUX
        )

        cov_records = []

        for parameter in cov_parameters:

            if parameter not in df.columns:
                continue

            for (
                line,
                machine
            ), machine_df in df.groupby(
                ["line", "machine_name"]
            ):

                machine_df = (
                    machine_df
                    .sort_values("eff_time")
                    .drop_duplicates(
                        subset=["Panel_SN"],
                        keep="first"
                    )
                    .reset_index(drop=True)
                )

                if len(machine_df) < 50:
                    continue

                values = pd.to_numeric(
                    machine_df[parameter],
                    errors="coerce"
                )

                for start in range(
                    len(machine_df) - 49
                ):

                    window_values = (
                        values.iloc[
                            start:
                            start + 50
                        ]
                        .dropna()
                    )

                    if len(window_values) < 2:
                        continue

                    cov_value = (
                        OutOfSpecListService
                        .calculate_cov(
                            window_values
                        )
                    )

                    if (
                        not OutOfSpecListService
                        .is_cov_fail(
                            parameter,
                            cov_value
                        )
                    ):
                        continue

                    cov_records.append(
                        {
                            "line": line,
                            "machine": machine,
                            "parameter": parameter,
                            "cov": cov_value
                        }
                    )

        if not cov_records:
            return []

        cov_df = pd.DataFrame(
            cov_records
        )

        for (
            line,
            parameter
        ), parameter_df in cov_df.groupby(
            [
                "line",
                "parameter"
            ]
        ):

            machine_rank = (
                parameter_df
                .groupby("machine")
                .size()
                .sort_values(
                    ascending=False
                )
                .head(3)
            )

            top3_machine = list(
                machine_rank.index
            )

            rows.append(
                {
                    "line": line,

                    "parameter": parameter,

                    "top3_machine":
                        ", ".join(
                            top3_machine
                        ),

                    "value": round(
                        parameter_df["cov"]
                        .mean(),
                        4
                    ),

                    "usl":
                        OutOfSpecListService
                        .get_cov_limit(
                            parameter
                        ),

                    "lsl": 0,

                    "number_of_strip":
                        len(parameter_df),

                    "status":
                        "Out COV"
                }
            )

        return rows

    @staticmethod
    def get_list(
        process: str,
        period_type: str,
        period_value: str,
        scope: str = "All"
    ):

        table_name = (
            OutOfSpecListRepository
            .get_table(process)
        )

        where_clause = (
            OutOfSpecListService
            .build_where_clause(
                process,
                period_type,
                period_value,
                scope
            )
        )

        df = (
            OutOfSpecListRepository
            .get_raw_data(
                table_name,
                where_clause
            )
        )

        parameter_defs = (
            OutOfSpecListRepository
            .get_parameter_definitions(
                table_name
            )
        )

        spec_rows = (
            OutOfSpecListService
            .build_out_spec_rows(
                df,
                parameter_defs
            )
        )

        cov_rows = (
            OutOfSpecListService
            .build_out_cov_rows(
                df,
                process
            )
        )

        all_rows = (
            spec_rows
            + cov_rows
        )

        all_rows.sort(
            key=lambda x:
                x["number_of_strip"],
            reverse=True
        )

        return {
            "process": process,
            "period_type": period_type,
            "period_value": period_value,
            "scope": scope,
            "total": len(all_rows),
            "items": all_rows
        }
